import tkinter as tk

janela = tk.Tk()
janela.title("Test")
janela.geometry("300x300")
texto = tk.Label(
    janela,
    text="Digite a sua senha",
    fg="black",
    bg="white",
    font=("Arial", 12)
)

texto.pack(pady=20)

barra = tk.Entry(janela)
barra.pack(pady=10)

def confirma_senha():
    senha_correta = "AdminTeste123"
    senha = barra.get()
    if senha == senha_correta:
        resultado = tk.Label(janela, text="Senha correta!", fg="green")
        
    else:
        resultado = tk.Label(janela, text="Senha incorreta!", fg="red")
    resultado.pack()

botao = tk.Button(janela, text="confirmar", command=confirma_senha)
botao.pack(pady=10)

janela.mainloop()
